import java.nio.charset.StandardCharsets
import java.net.URL
import org.apache.nifi.processor.io.OutputStreamCallback

def incomingFlowFile = session.get()
if (!incomingFlowFile) return

def urlString = "http://10.11.253.93:30117/stream/circuits"

try {
    def url = new URL(urlString)
    def connection = url.openConnection()
    connection.setRequestMethod("GET")

    def reader = connection.inputStream.newReader("UTF-8")

    for (int i = 0; i < 10; i++) {
        def line = reader.readLine()
        if (line == null) break

        if (!line.trim().isEmpty()) {

            log.info("DATA: " + line)

            def flowFile = session.create(incomingFlowFile)

            flowFile = session.write(flowFile, { out ->
                out.write(line.getBytes(StandardCharsets.UTF_8))
            } as OutputStreamCallback)

            session.transfer(flowFile, REL_SUCCESS)
        }
    }

    // pass original request forward
    session.transfer(incomingFlowFile, REL_SUCCESS)

} catch (Exception e) {
    log.error("Error reading stream: ${e.message}", e)
    session.transfer(incomingFlowFile, REL_FAILURE)
}


import java.nio.charset.StandardCharsets
import java.net.URL
import org.apache.nifi.processor.io.OutputStreamCallback

def flowFile = session.get()
if (!flowFile) return

def url = new URL("http://10.11.253.93:30117/stream/circuits")
def connection = url.openConnection()
connection.setRequestMethod("GET")

def reader = connection.inputStream.newReader("UTF-8")

def line = reader.readLine()

if (line && !line.trim().isEmpty()) {

    log.info("STREAM DATA: " + line)

    flowFile = session.write(flowFile, { out ->
        out.write(line.getBytes(StandardCharsets.UTF_8))
    } as OutputStreamCallback)

    session.transfer(flowFile, REL_SUCCESS)

} else {
    session.transfer(flowFile, REL_FAILURE)
}
