<!--
  Fragment Operations: circuit operations on dynamic fragments (dynamicFragment: true),
  grouped per fragment.

    create  (fragment buildSteps)    → createFragment(circuitId, fragmentTemplateId, request)
    delete  (fragment teardownSteps) → deleteFragment(circuitId, fragmentInstanceId, request)
    operate (neither)                → not wired yet (TODO)

  Each group shows how many instances the circuit has (no min/max limits enforced).
  Remove is disabled only when there are no instances.
  Delete and operate ask which instance first.
-->
<template>
  <div id="dynamic-fragment-operations">

        <span class="font-weight-bold">{{ group.name }}</span>
        <v-spacer />
        <v-chip size="small" color="primary" variant="flat">
          {{ instanceCount(group) }} {{ instanceCount(group) === 1 ? 'instance' : 'instances' }}
        </v-chip>
      </div>

      <v-data-table-virtual
        :id="`dynamic-fragment-operations-table-${group.fragmentId}`"

        item-value="key"
        density="compact"
      >
        <template #item.data-table-expand="{ item }">
          <div class="py-1">
            <v-btn

              prepend-icon="mdi-cog"

                  density="compact"
                  class="mb-2"

<script setup lang="ts">
import { useNotificationStore } from '@babcock/common-components'
import { computed, ref, toRaw, watch } from 'vue'

import { FormData } from '@/types/FormData'
import { createFragment, deleteFragment } from '../../api/requests'
import { OperationResult } from '../../types/OperationResult'
import { createCircuitOperationRequest } from '../../utils/CircuitOperationRequest'
import {
  DynamicFragmentGroup,
  DynamicFragmentOperation,
  getDynamicFragmentGroups,
  getFragmentInstances,
  getOperationAvailability,
} from '../../utils/templateFragments'
import PropertyFormInput from '../common/PropertyFormInput.vue'

const props = withDefaults(
  defineProps<{
    circuitId: string
    template: any // CircuitTemplate (fragmentTemplates / dynamicFragment may not be in the generated type)
    /** NEW: the circuit's current fragment instances (circuit.fragments) */
    fragments?: any[]
  }>(),
  { fragments: () => [] },
)

const emit = defineEmits<{ updated: [] }>()

interface Row extends DynamicFragmentOperation {
  /** unique per table row: `${fragmentId}:${operationId}` */
  key: string
  result: OperationResult
  loading: boolean
}

const notificationStore = useNotificationStore()

const headers = [
  { title: 'Operation name', key: 'name' },
  { title: 'Description', key: 'description' },
  { title: 'Actions', key: 'data-table-expand', width: '280px' },
  { title: 'Result', key: 'result' },
]

const circuitProperties = computed<Record<string, any>>(
  () => props.template?.circuitProperties ?? {},
)

const groups = ref<DynamicFragmentGroup[]>([])
const rows = ref<Record<string, Row>>({})
const formData = ref<Record<string, FormData>>({})
const forms = ref<Record<string, any>>({})
const expandedRows = ref<string[]>([])
const selectedInstance = ref<Record<string, string | undefined>>({})

// Rebuild when the template's fragments / operations change (including after it loads).
watch(
  () => getDynamicFragmentGroups(props.template),
  (newGroups) => {
    groups.value = newGroups
    const newRows: Record<string, Row> = {}
    const newForm: Record<string, FormData> = {}
    newGroups.forEach((g) =>
      g.operations.forEach((op) => {
        const key = `${g.fragmentId}:${op.id}`
        newRows[key] = { ...op, key, result: OperationResult.NOT_RUN, loading: false }
        newForm[key] = initialFormData(op.fragmentOperations)
      }),
    )
    rows.value = newRows
    formData.value = newForm
  },
  { immediate: true },
)

function rowsFor(group: DynamicFragmentGroup): Row[] {
  return group.operations.map((op) => rows.value[`${group.fragmentId}:${op.id}`]).filter(Boolean)
}

function instancesFor(group: DynamicFragmentGroup) {
  return getFragmentInstances(props.fragments, group)
}

function instanceCount(group: DynamicFragmentGroup): number {
  return instancesFor(group).length
}

function availability(item: Row) {
  const group = groups.value.find((g) => g.fragmentId === item.fragmentId)
  if (!group) return { allowed: false, reason: 'Unknown fragment' }
  if (item.kind === 'operate') {
    return { allowed: false, reason: 'Instance operations are not supported yet' }
  }
  return getOperationAvailability(item.kind, instanceCount(group))
}

/** Start each field from the template's values, keyed `${elementId}.${property|command}`. */
function initialFormData(fragmentOperations: Record<string, any>): FormData {
  const data: FormData = {}
  Object.values(fragmentOperations).forEach((elements: any) => {
    Object.entries(elements ?? {}).forEach(([elementId, statements]: [string, any]) => {
      ;(statements ?? []).forEach((statement: any) => {
        if ('property' in statement) {
          data[`${elementId}.${statement.property}`] = clone(statement.value)
        } else if ('command' in statement) {
          data[`${elementId}.${statement.command}`] = clone(statement.parameters)
        }
      })
    })
  })
  return data
}

/** Plain copy of template values (structuredClone throws on Vue reactive proxies). */
function clone<T>(value: T): T {
  return value === undefined ? value : JSON.parse(JSON.stringify(toRaw(value)))
}

function toggleExpandedRow(key: string) {
  expandedRows.value = expandedRows.value.includes(key)
    ? expandedRows.value.filter((r) => r !== key)
    : [...expandedRows.value, key]
}

async function run(item: Row, group: DynamicFragmentGroup) {
  if (!availability(item).allowed) return

  const { valid } = (await forms.value[item.key]?.validate()) ?? { valid: true }
  if (!valid) return

  item.loading = true
  try {
    // { fragment: { element: { … } } }, same shape as circuit operations
    const request = createCircuitOperationRequest(item.fragmentOperations, formData.value[item.key])

    let status: number
    if (item.kind === 'create') {
      status = await createFragment(props.circuitId, group.fragmentId, request)
    } else {
      const instanceId = selectedInstance.value[item.key]
      if (!instanceId) return
      status = await deleteFragment(props.circuitId, instanceId, request)
    }

    item.result = status >= 200 && status < 300 ? OperationResult.SUCCESS : OperationResult.FAILURE

    if (item.result === OperationResult.SUCCESS) {
      toggleExpandedRow(item.key)
      selectedInstance.value[item.key] = undefined
      emit('updated') // CircuitDetails reloads the circuit → fragments / counts refresh
    }
  } catch (error: any) {
    item.result = OperationResult.FAILURE
    notificationStore.error(`${item.name} failed: ${error.message}`)
    console.error(error)
  } finally {
    item.loading = false
  }
}

// Same chip look as CircuitOperations.vue. Swap for its helpers if they're exported.
function getResultColor(result: OperationResult): string {
  switch (result) {
    case OperationResult.SUCCESS:
      return 'success'
    case OperationResult.FAILURE:
      return 'error'
    default:
      return ''
  }
}

function getResultIcon(result: OperationResult): string {
  switch (result) {
    case OperationResult.SUCCESS:
      return 'mdi-check-circle'
    case OperationResult.FAILURE:
      return 'mdi-close-circle'
    default:
      return ''
  }
}
</script>

import type {
  CircuitFragmentTemplate,
  CircuitOperation,
  CircuitTemplate,
} from '@/api/generated/circuit-manager'

/**
 * Dynamic fragments and their circuit operations.
 *
 * Template (read-only, never modified here):
 *   fragmentTemplates: [ { id: 'enmf', dynamicFragment: true, multiplicity: { min, max },
 *                          buildSteps: [{ operationId: 'add-operator' }],
 *                          teardownSteps: [{ operationId: 'delete-operator' }], … }, … ]
 *   circuitOperations:
 *     add-operator:    { fragmentOperations: { enmf: { … } } }   ← create
 *     delete-operator: { fragmentOperations: { enmf: { … } } }   ← delete
 */

/**
 * create  = listed in the fragment's buildSteps    → adds a new instance (no instance picker)
 * delete  = listed in the fragment's teardownSteps → removes an existing instance
 * operate = targets the fragment but is neither    → acts on an existing instance
 */
export type DynamicOperationKind = 'create' | 'delete' | 'operate'

export interface DynamicFragmentOperation {
  /** circuit operation id, e.g. 'add-operator' */
  id: string
  name: string
  description: string
  /** dynamic fragment template it acts on, e.g. 'enmf' */
  fragmentId: string
  kind: DynamicOperationKind
  fragmentOperations: Record<string, any>
}

export interface DynamicFragmentGroup {
  /** fragment template id, e.g. 'enmf' */
  fragmentId: string
  /** display name, e.g. 'ENMF' */
  name: string
  min: number
  /** undefined or -1 = no maximum */
  max?: number
  operations: DynamicFragmentOperation[]
}

/** A fragment template is dynamic when the template says `dynamicFragment: true`. */
export function isDynamicFragment(fragment?: CircuitFragmentTemplate): boolean {
  return (fragment as any)?.dynamicFragment === true
}

function dynamicFragmentTemplates(template?: CircuitTemplate): any[] {
  return ((template as any)?.fragmentTemplates ?? []).filter(
    (f: CircuitFragmentTemplate) => f.id && isDynamicFragment(f),
  )
}

/** ids of dynamic fragment templates (e.g. ['enmf']). */
export function getDynamicFragmentIds(template?: CircuitTemplate): Set<string> {
  return new Set(dynamicFragmentTemplates(template).map((f) => f.id as string))
}

/** True if the circuit operation runs against a dynamic fragment. */
export function isDynamicFragmentOperation(
  operation: CircuitOperation | undefined,
  dynamicFragmentIds: Set<string>,
): boolean {
  // keys of fragmentOperations are fragment template ids (e.g. 'enmf')
  return Object.keys((operation as any)?.fragmentOperations ?? {}).some((fragmentId) =>
    dynamicFragmentIds.has(fragmentId),
  )
}

/** Show the Fragment Operations section only if at least one operation is dynamic. */
export function hasDynamicFragmentOperations(template?: CircuitTemplate): boolean {
  return getDynamicFragmentGroups(template).some((g) => g.operations.length > 0)
}

/**
 * One group per dynamic fragment, with the operations that act on it.
 *
 * Ownership: an operation belongs to the dynamic fragment whose buildSteps / teardownSteps
 * name it. If no steps name it, it falls back to the first dynamic fragment in its
 * fragmentOperations (kind 'operate'). Each operation appears in exactly one group.
 */
export function getDynamicFragmentGroups(template?: CircuitTemplate): DynamicFragmentGroup[] {
  const fragments = dynamicFragmentTemplates(template)
  const ops: Record<string, any> = (template as any)?.circuitOperations ?? {}
  const byId = new Map(fragments.map((f) => [f.id as string, f]))
  const groups = new Map<string, DynamicFragmentGroup>(
    fragments.map((f) => [
      f.id as string,
      {
        fragmentId: f.id as string,
        name: f.name || f.id,
        min: f.multiplicity?.min ?? 0,
        max: f.multiplicity?.max ?? undefined,
        operations: [],
      },
    ]),
  )

  const stepOwner = (operationId: string, steps: 'buildSteps' | 'teardownSteps') =>
    fragments.find((f) => (f[steps] ?? []).some((s: any) => s.operationId === operationId))

  Object.entries(ops).forEach(([id, op]) => {
    const builder = stepOwner(id, 'buildSteps')
    const tearer = stepOwner(id, 'teardownSteps')

    let fragmentId: string | undefined
    let kind: DynamicOperationKind
    if (builder) {
      fragmentId = builder.id
      kind = 'create'
    } else if (tearer) {
      fragmentId = tearer.id
      kind = 'delete'
    } else {
      fragmentId = Object.keys(op?.fragmentOperations ?? {}).find((k) => byId.has(k))
      kind = 'operate'
    }
    if (!fragmentId) return

    groups.get(fragmentId)!.operations.push({
      id,
      name: op.name || id,
      description: op.description || '',
      fragmentId,
      kind,
      fragmentOperations: op.fragmentOperations ?? {},
    })
  })

  // create first, then operate, then delete, so Add sits above Remove
  const order: Record<DynamicOperationKind, number> = { create: 0, operate: 1, delete: 2 }
  groups.forEach((g) => g.operations.sort((a, b) => order[a.kind] - order[b.kind]))

  return [...groups.values()].filter((g) => g.operations.length > 0)
}

/** Flat list of all dynamic fragment operations (kept for existing callers). */
export function getDynamicFragmentOperations(
  template?: CircuitTemplate,
): DynamicFragmentOperation[] {
  return getDynamicFragmentGroups(template).flatMap((g) => g.operations)
}

// ---------- fragment instances on the circuit ----------

export interface FragmentInstance {
  /** id of the fragment instance on the circuit (sent to delete / operate) */
  id: string
  /** label shown in the instance dropdown */
  label: string
}

/**
 * Template id of a fragment instance on the circuit.
 * TODO: confirm the field name on the generated `Fragment` type
 * (backend uses getFragmentByTemplateId, so it's likely `templateId` or `fragmentTemplateId`).
 */
export function getInstanceTemplateId(fragment: any): string | undefined {
  return fragment?.templateId ?? fragment?.fragmentTemplateId ?? fragment?.template
}

/** Instances of one dynamic fragment template currently on the circuit. */
export function getFragmentInstances(
  circuitFragments: any[] | undefined,
  group: Pick<DynamicFragmentGroup, 'fragmentId' | 'name'>,
): FragmentInstance[] {
  return (circuitFragments ?? [])
    .filter((f) => getInstanceTemplateId(f) === group.fragmentId)
    .map((f, i) => ({
      id: f.id,
      label: f.name ? `${f.name}` : `${group.name} #${i + 1}`,
    }))
}

/**
 * Whether an operation can run. No min/max limits are enforced; delete / operate only need
 * at least one instance to pick from.
 */
export function getOperationAvailability(
  kind: DynamicOperationKind,
  instanceCount: number,
): { allowed: boolean; reason?: string } {
  if (kind !== 'create' && instanceCount === 0) {
    return { allowed: false, reason: kind === 'delete' ? 'Nothing to remove yet' : 'No instances yet' }
  }
  return { allowed: true }
}

/** "min 0 · no max" */
export function formatLimits(group: Pick<DynamicFragmentGroup, 'min' | 'max'>): string {
  const max = group.max == null || group.max === -1 ? 'no max' : `max ${group.max}`
  return `min ${group.min} · ${max}`
}
