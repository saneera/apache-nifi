@Service
public class XmppCoreService {

    private XMPPTCPConnection connection;

    public void connect(String username, String password, String domain, String host, int port) throws Exception {
        XMPPTCPConnectionConfiguration config =
                XMPPTCPConnectionConfiguration.builder()
                        .setXmppDomain(domain)
                        .setHost(host)
                        .setPort(port)
                        .setSecurityMode(ConnectionConfiguration.SecurityMode.ifpossible)
                        .build();

        connection = new XMPPTCPConnection(config);
        connection.connect();
        connection.login(username, password);
    }

    public void disconnect() {
        if (connection != null) connection.disconnect();
    }

    public XMPPTCPConnection getConnection() {
        return connection;
    }

    public boolean isConnected() {
        return connection != null && connection.isAuthenticated();
    }
}

        submitForm.setAnswer("muc#roomconfig_allowinvites", true);

        muc.sendConfigurationForm(submitForm);

        MultiUserChat muc = getManager()
                .getMultiUserChat(JidCreate.entityBareFrom(roomJid));

        muc.join(Resourcepart.from(nickname));

        MultiUserChat muc = getManager()
                .getMultiUserChat(JidCreate.entityBareFrom(roomJid));

public List<RoomDto> listRooms() throws Exception {

        MultiUserChatManager manager =
                MultiUserChatManager.getInstanceFor(smackService.getConnection());

        String mucDomain = "conference." + properties.getDomain();

        DomainBareJid serviceJid = JidCreate.domainBareFrom(mucDomain);

        Collection<HostedRoom> hostedRooms =
                manager.getRoomsHostedBy(serviceJid);

        List<RoomDto> response = new ArrayList<>();

        for (HostedRoom hosted : hostedRooms) {

            try {

                EntityBareJid roomJid =
                        JidCreate.entityBareFrom(hosted.getJid().toString());

                MultiUserChat muc = manager.getMultiUserChat(roomJid);

                Form form = muc.getConfigurationForm();

                RoomDto dto = RoomDto.builder()
                        .jid(hosted.getJid().toString())
                        .name(hosted.getName())
                        .description(hosted.getName()) // fallback if desc missing
                        .persistent(getBoolean(form, "muc#roomconfig_persistentroom"))
                        .isPublic(getBoolean(form, "muc#roomconfig_publicroom"))
                        .membersOnly(getBoolean(form, "muc#roomconfig_membersonly"))
                        .passwordProtected(
                                getBoolean(form, "muc#roomconfig_passwordprotectedroom"))
                        .moderated(getBoolean(form, "muc#roomconfig_moderatedroom"))
                        .occupants(hosted.getOccupantsCount())
                        .maxUsers(getInt(form, "muc#roomconfig_maxusers", 50))
                        .createdAt(Instant.now().toString()) // if not stored elsewhere
                        .build();

                response.add(dto);

            } catch (Exception ex) {
                log.warn("Unable to read room {}", hosted.getJid(), ex);
            }
        }

        return response;
    }

    public RoomDto createRoom(RoomCreateRequest request) throws Exception {

            MultiUserChatManager manager =
                    MultiUserChatManager.getInstanceFor(smackService.getConnection());

            String roomJidString =
                    request.getRoomName() + "@conference." + properties.getDomain();

            EntityBareJid roomJid =
                    JidCreate.entityBareFrom(roomJidString);

            MultiUserChat muc = manager.getMultiUserChat(roomJid);

            // create room
            muc.create(Resourcepart.from(request.getOwnerNick()));

            // load config form
            Form configForm = muc.getConfigurationForm();
            Form submitForm = configForm.createAnswerForm();

            // copy defaults for hidden fields
            for (FormField field : configForm.getFields()) {
                if (!FormField.Type.hidden.equals(field.getType())
                        || field.getVariable() == null) {
                    continue;
                }
                submitForm.setDefaultAnswer(field.getVariable());
            }

            // set room config
            submitForm.setAnswer("muc#roomconfig_roomname", request.getRoomName());
            submitForm.setAnswer("muc#roomconfig_roomdesc", request.getDescription());
            submitForm.setAnswer("muc#roomconfig_maxusers",
                    String.valueOf(request.getMaxUsers()));

            submitForm.setAnswer("muc#roomconfig_persistentroom",
                    request.isPersistent());

            submitForm.setAnswer("muc#roomconfig_publicroom",
                    request.isPublic());

            submitForm.setAnswer("muc#roomconfig_membersonly",
                    request.isMembersOnly());

            submitForm.setAnswer("muc#roomconfig_moderatedroom",
                    request.isModerated());

            submitForm.setAnswer("muc#roomconfig_passwordprotectedroom",
                    request.isPasswordProtected());

            if (request.isPasswordProtected()
                    && request.getPassword() != null
                    && !request.getPassword().isBlank()) {

                submitForm.setAnswer("muc#roomconfig_roomsecret",
                        request.getPassword());
            }

            muc.sendConfigurationForm(submitForm);

            return RoomDto.builder()
                    .jid(roomJidString)
                    .name(request.getRoomName())
                    .description(request.getDescription())
                    .persistent(request.isPersistent())
                    .isPublic(request.isPublic())
                    .membersOnly(request.isMembersOnly())
                    .passwordProtected(request.isPasswordProtected())
                    .moderated(request.isModerated())
                    .occupants(1) // creator joined
                    .maxUsers(request.getMaxUsers())
                    .createdAt(Instant.now().toString())
                    .build();
        }

    private boolean getBoolean(Form form, String field) {
        try {
            String value = form.getField(field).getFirstValue();
            return "1".equals(value) || "true".equalsIgnoreCase(value);
        } catch (Exception ex) {
            return false;
        }
    }

    private int getInt(Form form, String field, int defaultValue) {
        try {
            return Integer.parseInt(form.getField(field).getFirstValue());
        } catch (Exception ex) {
            return defaultValue;
        }
    }
